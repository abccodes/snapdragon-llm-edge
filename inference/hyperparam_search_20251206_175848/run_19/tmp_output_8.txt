```python
# Importing necessary libraries
import pandas as pd

# Load the data from a CSV file (Assuming the CSV file contains information about different states and their peach production)
data = pd.read_csv('peach_production.csv')

# Extracting the state names and peach production values
state_names = data['State'].unique()
production_values = data['Production'].values

# Creating a dictionary to store the state with the highest production value
highest_production_state = {}

# Iterating through each state and its corresponding production value
for i in range(len(state_names)):
    if production_values[i] > highest_production_state.get('Highest Production', 0):
        highest_production_state['Highest Production'] = production_values[i]
        highest_production_state['State'] = state_names[i]

# Printing the result
print("The U.S. state that produces the most peaches is:", highest_production_state['State'])
```
