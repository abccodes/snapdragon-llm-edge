```python
# Importing necessary libraries
import pandas as pd

# Reading data from CSV file
data = pd.read_csv('peach_production.csv')

# Getting column names of DataFrame
column_names = list(data.columns)

# Filtering the data to get only peach production states
states_with_peaches = [state for state in column_names if 'Peach' in state]

# Printing the name of the state that produces the most peaches
max_peach_production_state = max(states_with_peaches, key=data['Production'])
print('The U.S. state that produces the most peches is:', max_peach_production_state)
```
