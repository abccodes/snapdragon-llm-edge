import requests

# Define the URL of the API that provides data on peach production in the United States
url = 'https://api.farmdocdata.com/peach_production'

# Send a GET request to the API and store the response in a variable
response = requests.get(url)

# Convert the response to JSON format and store it in a variable
data = response.json()

# Iterate through the data and find the state with the highest peach production
max_production = 0
max_production_state = ''

for state, production in data['states']:
    if production > max_production:
        max_production = production
        max_production_state = state

# Print the state with the highest peach production
print(f'The state that produces the most peaches in the United States is {max_production_state}.')
