```python
# Importing necessary libraries
import pandas as pd

# Loading the dataset from a CSV file
data = pd.read_csv('msg_data.csv')

# Grouping the data by 'Dose' column to calculate average response
grouped_data = data.groupby('Dose').mean()

# Printing the mean response for each dose
for dose in grouped_data.groups.keys():
    print(f"Mean Response at Dose: {dose} mg/kg: {grouped_data[grouped_data['Response'].sum() / len(grouped_data[grouped_data['Response']]]):.2f}")
```
